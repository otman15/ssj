48public class KorobovLattice extends Rank1Lattice {
49 protected int genA; // multiplier a
50 private int genT; // shift t
51
52 private void initN(int n, int t) {
53 int a = (genA % n) + (genA < 0 ? n : 0);
54 genT = t;
55 long[] B = new long[dim];
56 B[0] = 1;
57 int j;
58 for (j = 0; j < t; j++)
59 B[0] *= a;
60 v[0] = B[0] * normFactor;
61 for (j = 1; j < dim; j++) {
62 B[j] = (a * B[j - 1]) % n;
63 v[j] = normFactor * B[j];
64 }
65 }
