193 public double skewness(boolean biasCorrection) {
194 if (numberObs() < 3) {
195 log.logp(Level.WARNING, "TallyStore", "skewness",
196 "Tally " + name + ": computing skewness() with only " + numObs + " observations");
197 return Double.NaN;
198 }
199 double n = numberObs(); // We want all computations to be made in double.
200 double avg = average();
201 double var = variance();
202 double sum = 0.0;
203 double[] obs = this.getArray();
204 double x;
205 for (int i = 0; i < n; i++) {
206 x = obs[i] - avg;
207 x = x * x * x;
208 sum += x;
209 }
210 if (biasCorrection)
211 sum = sum * n / ((n-1)*(n-2));
212 else {
213 // var *= (n-1) / n;
214 sum /= n;
215 }
216 sum /= (var * Math.sqrt(var));
217 return sum;
218 }

256 public double kurtosis(boolean biasCorrection, boolean excess) {
257 if (numberObs() < 4) {
258 log.logp(Level.WARNING, "TallyStore", "kurtosis",
259 "Tally " + name + ": computing kurtosis() with only " + numObs + " observations");
260 return Double.NaN;
261 }
262 double n = numberObs(); // We want all computations to be made in double.
263 double avg = average();
264 double var = variance();
265 double sum = 0.0;
266 double[] obs = this.getArray();
267 double x;
268 for (int i = 0; i < n; i++) {
269 x = obs[i] - avg;
270 x *= x;
271 sum += x * x;
272 }
273 sum /= (var * var);
274 // System.out.println("kurtosis2: n = " + n + ", sum = " + sum);
275 if (biasCorrection) {
276 sum = sum * n * (n+1) / ((n-1) * (n-2) * (n-3));
277 if (excess)
278 sum -= 3.0 * (n-1) * (n-1) / ((n-2) * (n-3));
279 }
280 else {
281 // sum = sum * n / ((n-1) * (n-1)); // In case we want var *= (n-1) / n;
282 sum /= n;
283 if (excess) sum -= 3;
284 }
285 // System.out.println(" sum = " + sum);
286 return sum;
287 }

370 public TallyStore aggregate(int gsize) {
371 int numObs = this.numberObs();
372 int numGroups = numObs / gsize;
373 double[] obs = this.getArray();
374 double sum;
375 TallyStore t = new TallyStore(numGroups);
376 for (int i = 0; i < numGroups; i++) {
377 sum = 0.0;
378 for (int j = 0; j < gsize; j++)
379 sum += obs[gsize * i + j];
380 sum /= gsize;
381 t.add(sum);
382 }
383 // This is if gsize does not divide numObs.
384 int rest = numObs - numGroups * gsize;
385 if (rest > 0) {
386 sum = 0.0;
387 for (int j = 0; j < rest; j++)
388 sum += obs[gsize * numGroups + j];
389 sum /= rest;
390 t.add(sum);
391 }
392 return t;
393 }
