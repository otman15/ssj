68 public void randomize(PointSet p) {
69 if (p instanceof DigitalNet) {
70 ((DigitalNet) p).leftMatrixScramble(stream);
71 ((DigitalNet) p).addRandomShift(stream);
72 // System.out.println("The points immediately after LMS + shift");
73 // System.out.println(p.formatPoints());
74 }
75 else if (p instanceof ContainerPointSet) {
76 randomize(((ContainerPointSet) p).getOriginalPointSet());
77 }
78 else if (p instanceof CachedPointSet) {
79 randomize(((CachedPointSet) p).getParentPointSet());
80 }
81 else
82 throw new IllegalArgumentException("LMScrambleShift" + " can only randomize a DigitalNet");
83 }
84
88 public String toString() {
89 return "Left matrix scramble + random digital shift";
