77 public long[][][] generatorMatricesToBitFormat() {
78 int r, c, j; // Row r, column c, dimension j.
79 long[][][] bitMatrices = new long[dim][numRows][numCols];
80 for (j = 0; j < dim; j++) {
81 for (c = 0; c < numCols; c++) {
82 long column = genMat[j * numCols + c]; // This column as an integer.
83 column >>= outDigits - numRows; // outDigits (or w) is used here.
84 for (r = numRows - 1; r >= 0; r--) {
85 bitMatrices[j][r][c] = (column & 1L);
86 column >>= 1;
87 }
88 }
89 }
90 return bitMatrices;
91 }

158 public double getCoordinate(int i, int j) {
159 long res;
160 int pos = 0;
161 int grayCode = i ^ (i >> 1);
162
163 if (digitalShift == null)
164 res = 0;
165 else
166 res = digitalShift[j];
167 while ((grayCode >> pos) != 0) {
168 if (((grayCode >> pos) & 1) != 0)
169 res ^= genMat[j * numCols + pos];
170 pos++;
171 }
172 if (digitalShift != null)
173 return res * normFactor + EpsilonHalf;
174 else
175 return res * normFactor;
176 }

178 public double getCoordinateNoGray(int i, int j) {
179 long res;
180 if (digitalShift == null)
181 res = 0;
182 else
183 res = digitalShift[j];
184 int pos = 0; // Position of the bit that is examined.
185 // Add (xor) the columns of C_j for which the corresponding bit of i is 1.
186 // Least significant bit of i goes with first column of C_j.
187 // The first row of C_j is for the most significant bit of output, u_{i,j,1}.
188 while ((i >> pos) != 0) {
189 if ((((i >> pos) & 1) != 0) && (pos < numCols))
190 res ^= genMat[j * numCols + pos];
191 pos++;
192 }
193 if (digitalShift != null)
194 // This EpsilonHalf must be explained in the doc.
195 // It prevents the output of 0.
196 // normFactor is 2^{-outDigits}.
197 return res * normFactor + EpsilonHalf;
198 else
199 return res * normFactor;
200 }

225 if (null == stream)
226 throw new IllegalArgumentException(PrintfFormat.NEWLINE + " Calling addRandomShift with null stream");
227 if (0 == d2)
228 d2 = Math.max(1, dim);
229 if (digitalShift == null) {
230 digitalShift = new long[d2];
231 capacityShift = d2;
232 } else if (d2 > capacityShift) {
233 int d3 = Math.max(4, capacityShift);
234 while (d2 > d3)
235 d3 *= 2;
236 long[] temp = new long[d3];
237 capacityShift = d3;
238 for (int i = 0; i < d1; i++)

244 maxj = (1L << outDigits) - 1;

246 maxj = (1L << 63) - 1; // (long) 9223372036854775807; // 2^{63}-1
247 for (int i = d1; i < d2; i++)
248 digitalShift[i] = stream.nextLong(0, maxj); // ***************
249 dimShift = d2;
250 shiftStream = stream;
251 }

279 // Left-multiplies lower-triangular matrix Mj by original C_j,
280 // where original C_j is in originalMat and result is in genMat.
281 // Mj[d] is assumed to contain the d-th subdiagonal of matrix Mj,
282 // for d=0,...,w-1. Each subdiagonal is represented as a
283 // w-bit integer, whose most significant bits are those on the
284 // diagonal. For example, for d=w-3, the subdiagonal has 3 bits,
285 // say b1, b2, b3, and is represented by the integer
286 // Mj[w-3] = b1 * 2^{w-1} + b2 * 2^{w-2} + b3 * b^{w-3}.
287 //
288 private void leftMultiplyMat(int j, long[] Mj) {
289 int c, d; // Dimension j
290 long col; // column c for new C_j.
291 for (c = 0; c < numCols; c++) {
292 col = 0;
293 for (d = 0; d < outDigits; d++)
294 // Multiply subdiagonal d of M_j by column c of C_j, and xor.
295 col ^= (Mj[d] & originalMat[j * numCols + c]) >> d;
296 genMat[j * numCols + c] = col;
297 }
298 }

313 // Right-multiplies upper-triangular matrix Mj by original C_j,
314 // where original C_j is in originalMat and result is in genMat.
315 // Mj[d] is assumed to contain the d-th column of matrix Mj,
316 // for d=0,...,w-1. Each column is represented as a w-bit integer,
317 // whose most significant bits are those at index 0.
318 // For example, for d=2, the column has 3 bits, (the others are 0
319 // since under the diagonal) say b1, b2, b3, and is represented by
320 // the integer Mj[2] = b1 * 2^{w-1} + b2 * 2^{w-2} + b3 * b^{w-3}.
321 //
322 private void rightMultiplyMat(int j, long[] Mj) {
323 int c, r; // Dimension j
324 long col; // column c for new C_j.
325 long mask; // Bit of column Mj[c]
326
327 for (c = 0; c < numCols; c++) {
328 mask = (long)1 << outDigits - 1;
329 col = originalMat[j * numCols + c];
330 for (r = 0; r < c; r++) {
331 // If bit (outDigits - 1 - r) of Mj[c] is 1, add column r
332 if ((Mj[c] & mask) != 0)
333 col ^= originalMat[j * numCols + r];
334 mask >>= 1;
335 }
336 genMat[j * numCols + c] = col;
337 }
338 }

363 public void leftMatrixScramble (RandomStream stream, int numBits) {
364 int j, c, d; // dimension j, column c of original Cj, column d of new Cj.
365 int jk;
366 final long tworm1 = 1L << (numBits - 1); // 2^{numBits-1}
367 final int wmr = outDigits - numBits;
368 // If genMat contains the original gen. matrices, copy to originalMat.
369 // Normally, we do this only once!
370 if (originalMat == null) { // This is only if `originalMat` was never created.
371 // System.out.println("Copying genMat to originalMat \n");
372 originalMat = genMat;
373 genMat = new long[dim * numCols]; // Creates a new object, but only once.
374 }
375 for (d = 0; d < dim * numCols; d++) genMat[d] = 0;
376 for (j = 0; j < dim; j++) {
377 jk = j * numCols; // Number of columns to skip
378 for (c = 0; c < numRows; c++) {
379 // colc is column c of L_j, which has numRows columns.
380 long colc = (tworm1 + stream.nextLong(0, tworm1-1)) >> (c - wmr);
381 // System.out.println("colc = " + colc + "\n");
382 for (d = 0; d < numCols; d++) // Column d for coordinate j.
383 genMat[jk + d] ^= ((originalMat[jk + d] >> (outDigits-1-c)) & 1L) * colc;
384 }
385 }
386 // numRows = numBits; // The new generating matrices have r rows.
387 }

397 public void iBinomialMatrixScramble(RandomStream stream) {
398 int j, d; // Dimension j, subdiagonal d of M_j.
399 final long allOnes = ((long)1 << outDigits) - 1; // outDigits ones.
400 long lastRow; // Last row of M_j: w-1 random bits followed by 1.
401
402 // If genMat is original generator matrices, copy it to originalMat.
403 if (originalMat == null) {
404 originalMat = genMat;
405 genMat = new long[dim * numCols];
406 }
407 // Constructs the lower-triangular scrambling matrices M_j, w by w.
408 // scrambleMat[j][l] contains row l of M_j.
409 long[][] scrambleMat = new long[dim][outDigits];
410 for (j = 0; j < dim; j++) {
411 scrambleMat[j][0] = allOnes;
412 lastRow = stream.nextLong(0, allOnes) | 1;
413 for (d = 1; d < outDigits; d++)
414 // Subdiagonal d contains either all ones or all zeros.
415 if (((1 << d) & lastRow) == 0)
416 scrambleMat[j][d] = 0;
417 else
418 scrambleMat[j][d] = (allOnes >> d) << d;
419 }
420 for (j = 0; j < dim; j++)
421 leftMultiplyMat(j, scrambleMat[j]);
422 }

439 for (j = 0; j < dim; j++)
440 leftMultiplyMat(j, scrambleMat);

492 public void rightMatrixScramble(RandomStream stream) {
493 int j, c; // Dimension j, column c for new C_j.
494 if (originalMat == null) {
495 originalMat = genMat;
496 genMat = new long[dim * numCols];
497 }
498 // Generate an upper triangular matrix for Faure-Tezuka right-scramble.
499 // scrambleMat[c] contains column c of M.
500 long[] scrambleMat = new long[outDigits];
501 long boundLong = 0;
502 for (c = 0; c < numCols; c++) {
503 boundLong += (1 << c); // Integer repres. by string of c+1 ones.
504 scrambleMat[c] = (1 | stream.nextLong(0, boundLong)) << (outDigits - c - 1);
505 }
506 // Right-multiply the generator matrices by the scrambling matrix.
507 for (j = 0; j < dim; j++)
508 rightMultiplyMat(j, scrambleMat);
509 }

517 if (numBits < 1)
518 throw new IllegalArgumentException("numBits must be >= 1");
519 if (numBits > 31)

521 int maxj;
522 if (numBits < 31)
523 maxj = (1 << numBits) - 1;
524 else
525 maxj = 2147483647;
526 return stream.nextInt(0, maxj) << (31 - numBits);
527 }

571 public void nestedUniformScramble(RandomStream stream, double[][] output, int numBits) {
572 assert output.length == numPoints;
573 assert output.length > 0;
574 assert output[0].length == dim;
575 assert outDigits <= 31; // Maximum allowed in this version.
576 assert numBits <= 30; // Maximum allowed in this version.
577
578 if (numBits == 0)
579 numBits = outDigits-1;
580 // System.out.println("NUS: outDigits = " + outDigits);
581 double localNormFactor = 1.0 / Math.abs((double) (1L << (outDigits)));
582 System.out.println("NUS: outDigits = " + outDigits);
583 System.out.println("normFactor = " + localNormFactor);
584 int[] poslist = new int[2 * numPoints];
585 int[] bvlist = new int[2 * numPoints];
586 int[] counts = new int[256];
587 int[] binpos = new int[256];
588
589 for (int j = 0; j < dim; ++j) {
590 bvlist[0] = 0;
591 poslist[0] = 0;
592 for (int i = 1; i < numPoints; i++) {
593 // We use Gray code order (could be optional).
594 // We could have used a point set iterator here, but the iterator computes all
595 // coordinates at once and we need only one at a time.
596 int pos = 0;
597 int bv = 1;
598 while ((i & bv) == 0) {
599 pos++;
600 bv <<= 1;
601 }
602 bvlist[i] = bvlist[i - 1] ^ (int) (genMat[j * numCols + pos]); // >> (outDigits - 31));
603 poslist[i] = i;
604 }
605 // System.out.println("bvlist[numPoints-1] = " + bvlist[numPoints-1]);
606 for (int b = 0; b < 4; b++) {
607 for (int i = 0; i < 256; i++)
608 counts[i] = 0;
609 int m = (b % 2) * numPoints;
610 int bb = 8 * b;
611 int bv = 0xff << bb;
612 for (int i = 0; i < numPoints; i++)
613 counts[(bvlist[m + i] & bv) >>> bb]++;
614 binpos[0] = (1 - b % 2) * numPoints;
615 for (int i = 0; i < 255; i++)
616 binpos[i + 1] = binpos[i] + counts[i];
617 for (int i = 0; i < numPoints; i++) {
618 int pos = (bvlist[m + i] & bv) >>> bb;
619 int k = binpos[pos]++;
620 bvlist[k] = bvlist[m + i];
621 poslist[k] = poslist[m + i];
622 }
623 }
624 //System.out.println("bvlist[0] = " + Integer.toBinaryString(bvlist[0]));
625 //System.out.println("bvlist[numPoints-1] = " + bvlist[numPoints-1]);
626 // Note that there nothing random until here.
627 // The previous only initializes a data structure and it could be done only once
628 // for any given `k` and `genMat`. It can be recomputed only when `genMatNUS` changes.
629 int bv = randomBitVectorInt(stream, numBits); // Same bv for all points, but different bv2.
630 output[poslist[0]][j] = (bvlist[0] ^ bv) * localNormFactor + EpsilonHalf;
631 //System.out.println("bv = " + Integer.toBinaryString(bv) + ", log_2 bv = " + Math.log(bv)/Math.log(2.0));
632 //System.out.println("output[poslist[0]][j] = " + output[poslist[0]][j]);
633 //System.out.println("i = " + 0 + ", bvlist[0] = " + Long.toBinaryString(bvlist[0]));
634 for (int i = 1; i < numPoints; i++) {
635 // System.out.println("i = " + i + ", bvlist[i] = " + Long.toBinaryString(bvlist[i]));
636 int bv2 = bvlist[i - 1];
637 bv2 ^= bvlist[i];
638 bv2 = randomBitVectorInt(stream, numBits) & ((int) (1 << (int) Num.log2((double) bv2)) - 1);
639 bv ^= bv2;
640 // Here, the value of `poslist[i]` differs for the different coordinates j.
641 // This makes it much more difficult to avoid storing the points.
642 output[poslist[i]][j] = (bvlist[i] ^ bv) * localNormFactor + EpsilonHalf;
643 // System.out.println("bv2 = " + Integer.toBinaryString(bv2));
644 }
645 }

970 public void outputInterlace(long[][] points, double[][] interlacedPoints) {
971 assert points.length == numPoints;
972 assert points.length > 0;
973 assert points[0].length == dim;
974
975 assert interlacedPoints.length == numPoints;
976 assert interlacedPoints.length > 0;
977 assert interlacedPoints[0].length * interlacing == dim;
978
979 double longNormFactor = 1 / (Math.pow(2, 63));
980
981 for (int i = 0; i < numPoints; i++) {
982 for (int j = 0; j < dim / interlacing; ++j) {
983 long result = 0;
984 for (int idx = 0; idx < interlacing; ++idx) {
985 int coord = j * interlacing + idx;
986 int interlaced_pos = 62 - idx;
987 int original_pos = 30;
988 int mask = 1 << original_pos;
989 while (interlaced_pos >= 0 && original_pos >= 0) {
990 assert interlaced_pos >= original_pos;
991 result += (((long) (points[i][coord] & mask)) << (interlaced_pos - original_pos));
992 interlaced_pos -= interlacing;
993 original_pos -= 1;
994 mask >>= 1;
995 }
996 }
997 interlacedPoints[i][j] = result * longNormFactor + EpsilonHalf;
998 }
999 }
