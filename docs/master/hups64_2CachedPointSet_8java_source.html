77 public CachedPointSet(PointSet p, int fromPoint, int toPoint, int fromDim, int toDim) {
78 if (p.getNumPoints() < toPoint - fromPoint)
79 throw new IllegalArgumentException("Cannot cache more points than in point set p.");
80 if (p.getDimension() < toDim - fromDim)
81 throw new IllegalArgumentException("Attempt to cache points using more coordinates than the dimension.");
82 if (toPoint == Integer.MAX_VALUE)
83 throw new IllegalArgumentException("Cannot cache infinite number of points");
84 if (toDim == Integer.MAX_VALUE)
85 throw new IllegalArgumentException("Cannot cache infinite dimensional points");
86 this.fromPoint = fromPoint;
87 numPoints = toPoint - fromPoint;
88 this.fromDim = fromDim;
89 dim = toDim - fromDim;
90 this.P = p;
91 x = new double[numPoints][dim];
92 fillCache(fromDim, dim);
93 }
94
99 protected void fillCache(int fromDim, int dim) {
100 PointSetIterator itr = P.iterator();
101 if (fromPoint > 0)
102 itr.setCurPointIndex(fromPoint);
103 for (int i = 0; i < numPoints; i++)
104 itr.nextPoint(x[i], fromDim, dim);
105 }
