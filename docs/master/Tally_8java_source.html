115 public void add(double x) {
116 if (collect) {
117 if (x < minValue)
118 minValue = x;
119 if (x > maxValue)
120 maxValue = x;
121 numObs++;
122 // Updating algorithm from Knuth ed. 3, p. 232; see
123 // http://en.wikipedia.org/wiki/Algorithms_for_calculating_variance#cite_note-1
124 double y = x - curAverage;
125 curAverage += y / numObs;
126 curSum2 += y * (x - curAverage);
127 // On pourrait utiliser l'algorithme de Kahan pour une meilleure precision:
128 // (voir http://en.wikipedia.org/wiki/Kahan_summation_algorithm)
129 }
130 notifyListeners(x);
131 }

187 public double variance() {
188 // throws NumberObservationException {
189 // if (numObs < 2) throw NumberObservationException;
190 if (numObs < 2) {
191 // System.err.println (
192 // "******* Tally " + name + ": calling variance() with " + numObs +
193 // " Observation");
194 log.logp(Level.WARNING, "Tally", "variance",
195 "Tally " + name + ": calling variance() with only " + numObs + " observation");
196 return Double.NaN;
197 }
198 return curSum2 / (numObs - 1);
199 }

234 public void confidenceIntervalNormal(double level, double[] centerAndRadius) {
235 // Must return an array object, cannot return 2 doubles directly
236 double z;
237 if (numObs < 2)
238 throw new RuntimeException("Tally " + name + ": Calling confidenceIntervalStudent with < 2 Observations");
239 centerAndRadius[0] = average();
240 z = NormalDist.inverseF01(0.5 * (level + 1.0));
241 centerAndRadius[1] = z * Math.sqrt(variance() / (double) numObs);
242 }

267 public void confidenceIntervalStudent(double level, double[] centerAndRadius) {
268 // Must return an array object, cannot return 2 doubles directly
269 double t;
270 if (numObs < 2)
271 throw new RuntimeException("Tally " + name + ": Calling confidenceIntervalStudent with < 2 Observations");
272 centerAndRadius[0] = average();
273 t = StudentDist.inverseF(numObs - 1, 0.5 * (level + 1.0));
274 centerAndRadius[1] = t * Math.sqrt(variance() / (double) numObs);
275 }

370 public void confidenceIntervalVarianceChi2(double level, double[] interval) {
371 // Must return an array object, cannot return 2 doubles directly
372 if (numObs < 2)
373 throw new RuntimeException(
374 "Tally " + name + ": calling confidenceIntervalVarianceChi2 with < 2 observations");
375 double w = (numObs - 1) * variance();
376 double x2 = ChiSquareDist.inverseF(numObs - 1, 0.5 * (1.0 + level));
377 double x1 = ChiSquareDist.inverseF(numObs - 1, 0.5 * (1.0 - level));
378 interval[0] = w / x2;
379 interval[1] = w / x1;
380 }

423 public String report(double level, int d) {
424 PrintfFormat str = new PrintfFormat();
425 str.append("REPORT on Tally stat. collector ==> " + name);
426 str.append(
427 PrintfFormat.NEWLINE + " num. obs. min max average variance standard dev."
428 + PrintfFormat.NEWLINE);
429 str.append(7 + d, (int) numObs);
430 str.append(" ");
431 str.append(9 + d, d, d - 1, (double) minValue);
432 str.append(" ");
433 str.append(9 + d, d, d - 1, (double) maxValue);
434 str.append(" ");
435 str.append(9 + d, d, d - 1, (double) average());
436 str.append(" ");
437 str.append(9 + d, d, d - 1, (double) variance());
438 str.append(" ");
439 str.append(9 + d, d, d - 1, standardDeviation());
440 str.append(PrintfFormat.NEWLINE);
441
442 switch (confidenceInterval) {
443 case CI_NORMAL:
444 str.append(formatCINormal(level, d));
445 break;
446 case CI_STUDENT:
447 str.append(formatCIStudent(level, d));
448 break;
449 case CI_NONE:
450 break;
451 }
452
453 return str.toString();
454 }

480 public String shortReport() {
481 PrintfFormat pf = new PrintfFormat();
482 if (showNobs)
483 pf.append(-8, numberObs());
484 pf.append(9, 3, 2, min()).append(" ");
485 pf.append(9, 3, 2, max()).append(" ");
486 pf.append(10, 3, 2, average()).append(" ");
487 if (numberObs() >= 2) {
488 pf.append(10, 3, 2, variance()).append(" ");
489 pf.append(11, 3, 2, standardDeviation());
490 } else
491 pf.append(21, "---");
492
493 if (confidenceInterval != CIType.CI_NONE) {
494 double[] ci = new double[2];
495 switch (confidenceInterval) {
496 case CI_NORMAL:
497 confidenceIntervalNormal(level, ci);
498 break;
499 case CI_STUDENT:
500 confidenceIntervalStudent(level, ci);
501 break;
502 case CI_NONE:
503 break;
504 }
505 pf.append(" ").append((100 * level) + "% (");
506 pf.append(9, 3, 2, ci[0] - ci[1]).append(',');
507 pf.append(9, 3, 2, ci[0] + ci[1]).append(")");
508 }
509 return pf.toString();
510 }
