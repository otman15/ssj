64 public void randomize(PointSet p) {
65 if (p instanceof DigitalNet) {
66 ((DigitalNet) p).leftMatrixScramble(stream);
67 ((DigitalNet) p).addRandomShift(stream);
68 // System.out.println("The points immediately after LMS + shift");
69 // System.out.println(p.formatPoints());
70 }
71 else if (p instanceof ContainerPointSet) {
72 randomize(((ContainerPointSet) p).getOriginalPointSet());
73 }
74 else if (p instanceof CachedPointSet) {
75 randomize(((CachedPointSet) p).getParentPointSet());
76 }
77 else
78 throw new IllegalArgumentException("LMScrambleShift" + " can only randomize a DigitalNet");
79 }
