106 public void randomize(PointSet p) {
107 if (p instanceof CachedPointSet) {
108 CachedPointSet cp = (CachedPointSet) p;
109 if (cp.getParentPointSet() instanceof DigitalNetBase2) {
110 ((DigitalNetBase2) cp.getParentPointSet()).nestedUniformScramble(stream, cp.getArray(), numBits);
111 return;
112 }
113 }
114 else if (p instanceof ContainerPointSet) {
115 randomize(((ContainerPointSet) p).getOriginalPointSet());
116 }
117 else
118 throw new IllegalArgumentException(
119 "NestedUniformScrambling" + " can only randomize a CachedPointSet of a DigitalNetBase2");
120 }
