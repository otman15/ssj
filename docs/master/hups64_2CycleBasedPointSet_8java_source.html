59 public double getCoordinate(int i, int j) {
60 // Find cycle that contains point i, then index in cycle.
61 int l = 0; // Length of next cycle.
62 int n = 0; // Total length of cycles added so far.
63 int k;
64 for (k = 0; n <= i; k++)
65 n += l = ((AbstractList) cycles.get(k)).size();
66 AbstractList curCycle = (AbstractList) cycles.get(k - 1);
67 int coordinate = (i - n + l + j) % curCycle.size();
68// double[] curCycleD = ((DoubleArrayList) curCycle).elements();
69// return curCycleD[coordinate];
70 double x = ((DoubleArrayList) curCycle).get(coordinate);
71 return x;
72 }

91 public void addRandomShift(int d1, int d2, RandomStream stream) {
92 if (null == stream)
93 throw new IllegalArgumentException(PrintfFormat.NEWLINE + " Calling addRandomShift with null stream");
94 if (0 == d2)
95 d2 = Math.max(dim, 1);
96 if (shift == null) {
97 shift = new double[d2];
98 capacityShift = d2;
99 } else if (d2 > capacityShift) {
100 int d3 = Math.max(4, capacityShift);
101 while (d2 > d3)
102 d3 *= 2;
103 double[] temp = new double[d3];
104 capacityShift = d3;
105 for (int i = 0; i < d1; i++)
106 temp[i] = shift[i];
107 shift = temp;
108 }
109 dimShift = d2;
110 for (int i = d1; i < d2; i++)
111 shift[i] = stream.nextDouble();
112 shiftStream = stream;
113 }

118 public void clearRandomShift() {
119 super.clearRandomShift();
120 shift = null;
121 }

160 first = false;

162 sb.append(", ");
163 sb.append(cycle[e]);
164 }
165 sb.append(")");
166 }
167 return sb.toString();

215 public double nextCoordinate() {
216 // First, verify if there are still points....
217 if (getCurPointIndex() >= getNumPoints())
218 outOfBounds();
219 double x = curCycleD[curCoordInCycle];
220 if (shift != null) {
221 if (curCoordIndex >= dimShift) // Extend the shift.
222 addRandomShift(dimShift, curCoordIndex + 1, shiftStream);
223 x += shift[curCoordIndex];
224 if (x >= 1.0)
225 x -= 1.0;
226 if (x <= 0.0)
227 x = EpsilonHalf; // avoid x = 0
228 }
229 curCoordIndex++;
230 curCoordInCycle++;
231 if (curCoordInCycle >= curCycle.size())
232 curCoordInCycle = 0;
233 return x;
234 }

236 public void nextCoordinates(double p[], int dim) {
237 // First, verify if there are still points....
238 if (getCurPointIndex() >= getNumPoints())
239 outOfBounds();
240 if (curCoordIndex + dim >= dimShift)
241 addRandomShift(dimShift, curCoordIndex + dim + 1, shiftStream);
242 // int j = curCoordInCycle;
243 int maxj = curCycle.size();
244 double x;
245 for (int i = 0; i < dim; i++) {
246 x = curCycleD[curCoordInCycle++];
247 if (curCoordInCycle >= maxj)
248 curCoordInCycle = 0;
249 if (shift != null) {
250 x += shift[curCoordIndex + i];
251 if (x >= 1.0)
252 x -= 1.0;
253 if (x <= 0.0)
254 x = EpsilonHalf; // avoid x = 0
255 }
256 p[i] = x;
257 }
258 curCoordIndex += dim;
259 }
260
261 public void setCurPointIndex(int i) {
262 int l = 0;
263 int n = 0;
264 int j;
265 for (j = 0; n <= i; j++)
266 n += l = ((AbstractList) cycles.get(j)).size();
267 resetCurCycle(j - 1);
268 startPointInCycle = i - n + l;
269 curPointIndex = i;
270 curCoordIndex = 0;
271 curCoordInCycle = startPointInCycle;
272 }

277 curPointIndex = 0;
278 curCoordIndex = 0;
279 curCoordInCycle = 0;
280 }

282 public int resetToNextPoint() {
283 curPointIndex++;
284 startPointInCycle++;
285 if (startPointInCycle >= curCycle.size()) {
286 startPointInCycle = 0;
287 if (curCycleIndex < (numCycles - 1))
288 resetCurCycle(curCycleIndex + 1);
289 }
290 curCoordIndex = 0;
291 curCoordInCycle = startPointInCycle;
292 return curPointIndex;
293 }
294
295 public int nextPoint(double p[], int dim) {
296 // First, verify if there are still points....
297 if (getCurPointIndex() >= getNumPoints())
298 outOfBounds();
299 int j = startPointInCycle;
300 int maxj = curCycle.size() - 1;
301 for (int i = 0; i < dim; i++) {
302 p[i] = curCycleD[j];
303 if (j < maxj)
304 j++;
305 else
306 j = 0;
307 }
308 resetToNextPoint();
309 return curPointIndex;
310 }
