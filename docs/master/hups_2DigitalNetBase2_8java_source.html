64 public double getCoordinate(int i, int j) {
65 int res;
66 int pos = 0;
67 int grayCode = i ^ (i >> 1);
68 if (digitalShift == null)
69 res = 0;
70 else
71 res = digitalShift[j];
72 while ((grayCode >> pos) != 0) {
73 if (((grayCode >> pos) & 1) != 0)
74 res ^= genMat[j * numCols + pos];
75 pos++;
76 }
77 if (digitalShift != null)
78 return res * normFactor + EpsilonHalf;
79 else
80 return res * normFactor;
81 }
82
83 public double getCoordinateNoGray(int i, int j) {
84 int res;
85 if (digitalShift == null)
86 res = 0;
87 else
88 res = digitalShift[j];
89 int pos = 0; // Position of the bit that is examined.
90 // Add (xor) the columns of C_j for which the corresponding bit of i is 1.
91 // Least significant bit of i goes with first column of C_j.
92 // The first row of C_j is for the most significant bit of output, u_{i,j,1}.
93 while ((i >> pos) != 0) {
94 if ((((i >> pos) & 1) != 0) && (pos < numCols))
95 res ^= genMat[j * numCols + pos];
96 pos++;
97 }
98 if (digitalShift != null)
99 // This EpsilonHalf must be explained in the doc.
100 // It prevents the output of 0.
101 // normFactor is 2^{-outDigits}.
102 return res * normFactor + EpsilonHalf;
103 else
104 return res * normFactor;
105 }

171 // Left-multiplies lower-triangular matrix Mj by original C_j,
172 // where original C_j is in originalMat and result is in genMat.
173 // Mj[d] is assumed to contain the d-th subdiagonal of matrix Mj,
174 // for d=0,...,w-1. Each subdiagonal is represented as a
175 // w-bit integer, whose most significant bits are those on the
176 // diagonal. For example, for d=w-3, the subdiagonal has 3 bits,
177 // say b1, b2, b3, and is represented by the integer
178 // Mj[w-3] = b1 * 2^{w-1} + b2 * 2^{w-2} + b3 * b^{w-3}.
179 //
180 private void leftMultiplyMatSubdiag (int j, int[] Mj) {
181 int c, d, col; // Dimension j, column c for new C_j.
182 for (c = 0; c < numCols; c++) {
183 col = 0;
184 for (d = 0; d < outDigits; d++)
185 // Multiply subdiagonal d of M_j by column c of C_j, and xor.
186 col ^= (Mj[d] & originalMat[j * numCols + c]) >> d;
187 genMat[j * numCols + c] = col; // Column c for coordinate j.
188 }
189 }
190
191
192 // Right-multiplies upper-triangular matrix Mj by original C_j,
193 // where original C_j is in originalMat and result is in genMat.
194 // Mj[d] is assumed to contain the d-th column of matrix Mj,
195 // for d=0,...,w-1. Each column is represented as a w-bit integer,
196 // whose most significant bits are those at index 0.
197 // For example, for d=2, the column has 3 bits, (the others are 0
198 // since under the diagonal) say b1, b2, b3, and is represented by
199 // the integer Mj[2] = b1 * 2^{w-1} + b2 * 2^{w-2} + b3 * b^{w-3}.
200 //
201 private void rightMultiplyMat(int j, int[] Mj) {
202 int c, r, col; // Dimension j, column c for new C_j.
203 int mask; // Bit of column Mj[c]
204
205 for (c = 0; c < numCols; c++) {
206 mask = 1 << outDigits - 1;
207 col = originalMat[j * numCols + c];
208 for (r = 0; r < c; r++) {
209 // If bit (outDigits - 1 - r) of Mj[c] is 1, add column r
210 if ((Mj[c] & mask) != 0)
211 col ^= originalMat[j * numCols + r];
212 mask >>= 1;
213 }
214 genMat[j * numCols + c] = col;
215 }
216 }

312 // This is a version of LMS in which each subdiagonal of the lower-triangular
313 // scrambling matrix is represented as an integer.
314 // It is equivalent to `leftMatrixScrambleSubdiag (0, stream)`.
315 public void leftMatrixScrambleSubdiag (RandomStream stream) {
316 int j, d; // dimension j, subdiagonal d.
317 final int allOnes = (1 << outDigits) - 1; // outDigits ones.
318 System.out.println("allOnes = " + allOnes);
319
320 // If genMat contains the original gen. matrices, copy to originalMat.
321 // This is done only once.
322 if (originalMat == null) {
323 originalMat = genMat;
324 genMat = new int[dim * numCols];
325 System.out.println("new GenMat, numCols = " + numCols);
326 }
327 // Constructs the lower-triangular scrambling matrices M_j, w by w.
328 int[][] scrambleMat = new int[dim][outDigits]; // This creates a big object!
329 System.out.println("new scrambleMat, outDigits = " + outDigits);
330 for (j = 0; j < dim; j++) {
331 scrambleMat[j][0] = allOnes; // This is the diagonal of a w x w matrix.
332 for (d = 1; d < outDigits; d++)
333 // The d-th subdiagonal will contain w-d random bits.
334 // It is represented as a (w-d)-bit integer.
335 scrambleMat[j][d] = (stream.nextInt(0, allOnes >> d)) << d;
336 }
337 // Multiply M_j by the generator matrix C_j for each j.
338 for (j = 0; j < dim; j++)
339 leftMultiplyMatSubdiag(j, scrambleMat[j]);
340 }

345 int j, d; // dimension j, subdiagonal d.

356 // scrambleMat[j][l] contains row l in a single integer (binary repres.)

358 System.out.println("new scrambleMat, outDigits = " + outDigits);

373 public void iBinomialMatrixScramble(RandomStream stream) {
374 int j, d; // Dimension j, subdiagonal d of M_j.
375 final int allOnes = (1 << outDigits) - 1; // outDigits ones.
376 int lastRow; // Last row of M_j: w-1 random bits followed by 1.
377
378 // If genMat is original generator matrices, copy it to originalMat.
379 if (originalMat == null) {
380 originalMat = genMat;
381 genMat = new int[dim * numCols];
382 }
383 // Constructs the lower-triangular scrambling matrices M_j, w by w.
384 // scrambleMat[j][l] contains the subdiagonal l of M_j.
385 // *** Pierre: We should avoid creating and storing all these matrices. ****
386 int[][] scrambleMat = new int[dim][outDigits];
387 for (j = 0; j < dim; j++) {
388 scrambleMat[j][0] = allOnes;
389 lastRow = stream.nextInt(0, allOnes) | 1;
390 for (d = 1; d < outDigits; d++)
391 // Subdiagonal d contains either all ones or all zeros.
392 if (((1 << d) & lastRow) == 0)
393 scrambleMat[j][d] = 0;
394 else
395 scrambleMat[j][d] = (allOnes >> d) << d;

397 for (j = 0; j < dim; j++)
398 leftMultiplyMatSubdiag(j, scrambleMat[j]);

402 public void stripedMatrixScramble(RandomStream stream) {
403 int j, d; // dimension j, subdiagonal d of M_j.
404

407 originalMat = genMat;

415 scrambleMat[d] = (allOnes >> d) << d;
416 for (j = 0; j < dim; j++)
417 leftMultiplyMatSubdiag(j, scrambleMat);

421 public void rightMatrixScramble(RandomStream stream) {
422 int j, c; // Dimension j, column c for new C_j.
423 if (originalMat == null) {
424 originalMat = genMat;
425 genMat = new int[dim * numCols];
426 }
427 // Generate an upper triangular matrix for the Faure-Tezuka right-scramble.
428 // scrambleMat[c] contains column c of M.
429 int[] scrambleMat = new int[outDigits];
430 int boundInt = 0;
431 for (c = 0; c < numCols; c++) {
432 boundInt += (1 << c); // Integer repres. by string of c+1 ones.
433 scrambleMat[c] = (1 | stream.nextInt(0, boundInt)) << (outDigits - c - 1);
434 }
435 // Right-multiply the generator matrices by the scrambling matrix.
436 for (j = 0; j < dim; j++)
437 rightMultiplyMat(j, scrambleMat);
438 }

444 private int randomBitVector(RandomStream stream, int numBits) {
445 if (numBits < 1)
446 throw new IllegalArgumentException("numBits must be >= 1");
447 if (numBits > 31)
448 throw new IllegalArgumentException("numBits must be <= 31");
449 int maxj;
450 if (numBits < 31)
451 maxj = (1 << numBits) - 1;
452 else
453 maxj = 2147483647;
454 return stream.nextInt(0, maxj) << (31 - numBits);
455 }

504 public void nestedUniformScramble(RandomStream stream, int[][] output, int numBits) {
505 assert output.length == numPoints;
506 assert output.length > 0;
507 assert output[0].length == dim;
508 if (numBits == 0)
509 numBits = outDigits;
510 int[] poslist = new int[2 * numPoints];
511 int[] bvlist = new int[2 * numPoints];
512 int[] counts = new int[256];
513 int[] binpos = new int[256];
514
515 for (int j = 0; j < dim; ++j) {
516 bvlist[0] = 0;
517 poslist[0] = 0;
518 for (int i = 1; i < numPoints; i++) {
519 // We use Gray code order (could be optional).
520 // We could have used a point set iterator here, but the iterator computes all
521 // coordinates at once and we need only one at a time.
522 int pos = 0;
523 int bv = 1;
524 while ((i & bv) == 0) {
525 pos++;
526 bv <<= 1;
527 }
528 bvlist[i] = bvlist[i - 1] ^ genMat[j * numCols + pos];
529 poslist[i] = i;
530 }
531 for (int b = 0; b < 4; b++) {
532 for (int i = 0; i < 256; i++)
533 counts[i] = 0;
534 int m = (b % 2) * numPoints;
535 int bb = 8 * b;
536 int bv = 0xff << bb;
537 for (int i = 0; i < numPoints; i++)
538 counts[(bvlist[m + i] & bv) >>> bb]++;
539 binpos[0] = (1 - b % 2) * numPoints;
540 for (int i = 0; i < 255; i++)
541 binpos[i + 1] = binpos[i] + counts[i];
542 for (int i = 0; i < numPoints; i++) {
543 int pos = (bvlist[m + i] & bv) >>> bb;
544 int k = binpos[pos]++;
545 bvlist[k] = bvlist[m + i];
546 poslist[k] = poslist[m + i];
547 }
548 }
549 int bv = randomBitVector(stream, numBits);
550 output[poslist[0]][j] = bvlist[0] ^ bv;
551 for (int i = 1; i < numPoints; i++) {
552 int bv2 = bvlist[i - 1];
553 bv2 ^= bvlist[i];
554 bv2 = randomBitVector(stream, numBits) & ((int) (1 << (int) Num.log2((double) bv2)) - 1);
555 bv ^= bv2;
556 output[poslist[i]][j] = bvlist[i] ^ bv;
557 }
558 }
559 }

638 public int[][][] genMatricesToBitByBitFormat(){
639 int r, c, j; // Row r, column c, dimension j.
640 int[][][] bitMatrices = new int[dim][numRows][numCols];
641 for (j = 0; j < dim; j++) {
642 for (c = 0; c < numCols; c++) {
643 int column = genMat[j * numCols + c]; // This column as an integer.
644 column >>= outDigits - numRows; // outDigits (or w) is used here.
645 for (r = numRows - 1; r >= 0; r--) {
646 bitMatrices[j][r][c] = (column & 1);
647 column >>= 1;
648 }
649 }
650 }
651 return bitMatrices;
652 }

749 public void outputInterlace(int[][] points, double[][] interlacedPoints) {
750 assert points.length == numPoints;
751 assert points.length > 0;
752 assert points[0].length == dim;
753
754 assert interlacedPoints.length == numPoints;
755 assert interlacedPoints.length > 0;
756 assert interlacedPoints[0].length * interlacing == dim;
757
758 double longNormFactor = 1 / (Math.pow(2, 63));
759
760 for (int i = 0; i < numPoints; i++) {
761 for (int j = 0; j < dim / interlacing; ++j) {
762 long result = 0;
763 for (int idx = 0; idx < interlacing; ++idx) {
764 int coord = j * interlacing + idx;
765 int interlaced_pos = 62 - idx;
766 int original_pos = 30;
767 int mask = 1 << original_pos;
768 while (interlaced_pos >= 0 && original_pos >= 0) {
769 assert interlaced_pos >= original_pos;
770 result += (((long) (points[i][coord] & mask)) << (interlaced_pos - original_pos));
771 interlaced_pos -= interlacing;
772 original_pos -= 1;
773 mask >>= 1;
774 }
775 }
776 interlacedPoints[i][j] = result * longNormFactor + EpsilonHalf;
777 }
778 }

789 public DigitalNetBase2 matrixInterlace(int interlace, int newDim) {
790 assert interlace * newDim <= dim;
791 DigitalNetBase2 result = new DigitalNetBase2(); // Creates a new interlaced net object.
792 result.dim = newDim;
793 result.numPoints = numPoints;
794 result.outDigits = outDigits;
795 result.normFactor = normFactor;
796 result.numCols = numCols;
797 result.numRows = Math.min(outDigits, numRows * interlace);
798
799 int[][][] nonInterlacedMatrices = genMatricesToBitByBitFormat();
800 int[][][] interlacedMatrices = new int[result.dim][result.numRows][result.numCols];
801 int r, j; // Row r, dimension j.
802 for (j = 0; j < result.dim; ++j) {
803 for (r = 0; r < result.numRows; ++r) {
804 // This copies all the columns for row r.
805 interlacedMatrices[j][r] = nonInterlacedMatrices[j * interlace + r % interlace][r / interlace];
806 }
807 }
808 result.genMatricesFromBitByBitFormat(interlacedMatrices);
809 return result;
810 }

828 protected class DigitalNetBase2Iterator extends DigitalNetIterator {
829
830 // Coordinates of the current point stored (cached) as integers.
831 // Initially contains zeros, because first point is always zero.
832 // Incorporates the random shift, except for the first point.
833 // There is one more dimension for the points because of the
834 // shift iterators children of DigitalNetBase2Iterator.
835 // dimS = dim, except for the shift iterator children where
836 // dimS = dim + 1.
837 protected int dimS;
838
839 public DigitalNetBase2Iterator() {
840 super();
841 EpsilonHalf = 0.5 / Num.TWOEXP[outDigits];
842 cachedCurPoint = new int[dim + 1];
843 dimS = dim;
844 init2();
845 }
846
847 public void init() { // This empty method is necessary to overload
848 } // the init() of DigitalNetIterator
849
850 public void init2() { // See constructor
851 resetCurPointIndex();
852 }
853
854 // We want to avoid generating 0 or 1
855 public double nextDouble() {
856 return nextCoordinate();
857 }
858
859 public double nextCoordinate() {
860 if (curPointIndex >= numPoints || curCoordIndex >= dimS)
861 outOfBounds();
862 if (digitalShift == null)
863 return cachedCurPoint[curCoordIndex++] * normFactor;
864 else
865 return cachedCurPoint[curCoordIndex++] * normFactor + EpsilonHalf;
866 // *** Pierre: EpsilonHalf could be replaced by a variable `epsAdd` to avoid the "if".
867 }
868
869 protected void addShiftToCache() {
870 if (digitalShift == null)
871 for (int j = 0; j < dim; j++)
872 cachedCurPoint[j] = 0;
873 else {
874 if (dimShift < dimS)
875 addRandomShift(dimShift, dimS, shiftStream);
876 for (int j = 0; j < dim; j++)
877 cachedCurPoint[j] = digitalShift[j];
878 }
879 }
880
881 public void resetCurPointIndex() {
882 addShiftToCache();
883 curPointIndex = 0;
884 curCoordIndex = 0;
885 }
886
887 public void setCurPointIndex(int i) {
888 if (i == 0) {
889 resetCurPointIndex();
890 return;
891 }
892 // Out of order computation, must recompute the cached current
893 // point from scratch.
894 curPointIndex = i;
895 curCoordIndex = 0;
896 addShiftToCache();
897
898 int j;
899 int grayCode = i ^ (i >> 1);
900 int pos = 0; // Position of the bit that is examined.
901 while ((grayCode >> pos) != 0) {
902 if (((grayCode >> pos) & 1) != 0)
903 for (j = 0; j < dim; j++)
904 cachedCurPoint[j] ^= genMat[j * numCols + pos];
905 pos++;
906 }
907 }
908
909 public int resetToNextPoint() {
910 int pos = 0; // Will be position of change in Gray code,
911 // = pos. of first 0 in binary code of point index.
912 while (((curPointIndex >> pos) & 1) != 0)
913 pos++;
914 if (pos < numCols) {
915 for (int j = 0; j < dim; j++)
916 cachedCurPoint[j] ^= genMat[j * numCols + pos];
917 }
918 curCoordIndex = 0;
919 return ++curPointIndex;
920 }
921
922 public int nextPoint(double p[], int d) {
923 if (curPointIndex >= numPoints || d > dimS)
924 outOfBounds();
925 if (digitalShift == null) {
926 for (int j = 0; j < d; j++)
927 p[j] = cachedCurPoint[j] * normFactor;
928 } else {
929 for (int j = 0; j < d; j++)
930 p[j] = cachedCurPoint[j] * normFactor + EpsilonHalf;
931 }
932 return resetToNextPoint();
933 }
934 }
