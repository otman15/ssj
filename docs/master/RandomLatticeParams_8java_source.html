85 public void randomize(PointSet p) {
86 if (p instanceof Rank1Lattice) {
87 if (nPow2)
88 ((Rank1Lattice) p).setRandomAforPow2n(stream);
89 else {
90 if (nmax == 0)
91 ((Rank1Lattice) p).setRandomAforPrimen(stream);
92 else
93 ((Rank1Lattice) p).setRandomAandn(nmin, nmax, stream);
94 }
95 if (randShift)
96 ((Rank1Lattice) p).addRandomShift(stream);
97 } else if (p instanceof ContainerPointSet) {
98 randomize(((ContainerPointSet) p).getOriginalPointSet());
99 } else if (p instanceof CachedPointSet) {
100 randomize(((CachedPointSet) p).getParentPointSet());
101 } else
102 throw new IllegalArgumentException("RandomLatticeParams" + " can only randomize a Rank1Lattice");
103 }
