93 public void init(double a, double b, int numBins) {
94 /*
95 * The counters count[0] to count[s-1] contains the number of observations
96 * falling in the interval [a, b]. leftCount is the number of observations < a,
97 * and rightCount is the number of observations > b.
98 */
99 if (b <= a)
100 throw new IllegalArgumentException(" b <= a");
101 count = new int[numBins];
102 this.numBins = numBins;
103 m_h = (b - a) / (double) numBins;
104 m_a = a;
105 m_b = b;
106 init();
107 }

114 public void init() {
115 super.init();
116 leftCount = rightCount = 0;
117 for (int i = 0; i < numBins; i++)
118 count[i] = 0;
119 }

170 public void add(double x) {
171 super.add(x);
172 if (x < m_a)
173 ++leftCount;
174 else if (x > m_b)
175 ++rightCount;
176 else {
177 int i = (int) ((x - m_a) / m_h);
178 ++count[i];
179 }

186 public TallyHistogram trimHistogram() {
187 TallyHistogram image = (TallyHistogram) super.clone();
188 int i = 0;
189 int j = numBins - 1; // last bin in the initial histogram
190 int cpL = 0; // number of empty bins from left initialized to zero
191 int cpR = 0; // number of empty bins from right initialized to zero
192 while (count[i] == 0) {
193 i++;
194 cpL++;
195 }
196 while (count[j] == 0) {
197 j--;
198 cpR++;
199 }
200 int[] coco = new int[numBins - cpL - cpR];
201 System.arraycopy(count, i, coco, 0, j - i + 1);
202 image.count = coco;
203 image.m_h = m_h;
204 image.m_a = m_a + (cpL * m_h);
205 image.m_b = m_b - (cpR * m_h);
206 image.numBins = numBins - cpL - cpR;
207 image.leftCount = leftCount;
208 image.rightCount = rightCount;
209 return image;
210 }

220 public TallyHistogram addHistograms(TallyHistogram other) {
221 if (this.numBins != other.numBins)
222 throw new IllegalArgumentException("different number of bin in two histogram to merge");
223 TallyHistogram image = (TallyHistogram) super.clone();
224 int[] countNew = new int[numBins];
225 System.arraycopy(count, 0, countNew, 0, numBins);
226 int coOther[] = other.getCounters();
227 for (int i = 0; i < countNew.length; i++)
228 countNew[i] = countNew[i] + coOther[i];
229 image.count = countNew;
230 image.leftCount = leftCount + other.leftCount;
231 image.rightCount = rightCount + other.rightCount;
232 image.m_h = m_h;
233 image.m_a = m_a;
234 image.m_b = m_b;
235 image.numBins = numBins;
236 return image;

246 public TallyHistogram aggregateBins(int g) {
247 TallyHistogram image = (TallyHistogram) super.clone();
248 int numBinsNew = (int) Math.ceil((double) numBins / (double) g);
249 int[] countNew = new int[numBinsNew];
250 int b = 0;
251 for (int j = 0; j < numBinsNew - 1; j++) {
252 for (int i = b; i < b + g; i++)
253 countNew[j] += count[i];
254 b = b + g;
255 }
256 while (b < numBins - 1) {
257 countNew[numBinsNew - 1] += count[b];
258 b++;
259 }
260 image.count = countNew;
261 image.m_h = m_h * g;
262 image.m_a = m_a;
263 image.m_b = m_h * numBinsNew;
264 image.numBins = numBinsNew;
265 image.leftCount = leftCount;
266 image.rightCount = rightCount;
267 return image;
268 }

354 public String toString() {
355 StringBuffer sb = new StringBuffer();
356 sb.append("---------------------------------------" + PrintfFormat.NEWLINE);
357 sb.append(name + PrintfFormat.NEWLINE);
358 sb.append("Interval = [ " + m_a + ", " + m_b + " ]" + PrintfFormat.NEWLINE);
359 sb.append("Number of bins = " + numBins + PrintfFormat.NEWLINE);
360 sb.append(PrintfFormat.NEWLINE + "Counters = {" + PrintfFormat.NEWLINE);
361 sb.append(" (-inf, " + PrintfFormat.f(6, 3, m_a) + ") " + leftCount + PrintfFormat.NEWLINE);
362 for (int i = 0; i < numBins; i++) {
363 double a = m_a + (i - 1) * m_h;
364 double b = m_a + i * m_h;
365 sb.append(" (" + PrintfFormat.f(6, 3, a) + ", " + PrintfFormat.f(6, 3, b) + ") " + count[i]
366 + PrintfFormat.NEWLINE);
367 }
368 sb.append(" (" + PrintfFormat.f(6, 3, m_b) + ", inf) " + rightCount + PrintfFormat.NEWLINE);
369 sb.append("}" + PrintfFormat.NEWLINE);
370 return sb.toString();
